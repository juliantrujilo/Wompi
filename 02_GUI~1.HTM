Paso 1 · Modelos

Modelo Orden.js y modelo Transaccion.js

Separamos la transacción de Wompi del pedido. La transacción se crea antes del pago (con estado PENDING). La orden solo existe si el pago fue aprobado (APPROVED).

💡¿Por qué dos modelos?
La transacción representa el intento de pago. Si el pago falla → transacción DECLINED, la orden nunca se crea. Si el pago se aprueba → orden PAGO_CONFIRMADO. Esto evita tener órdenes "fantasma" con estado pendiente que nunca se pagaron.

Paso 2 · Ruta — firma

POST /api/pagos/firma — generar firma para el Widget

El Widget de Wompi no acepta montos sin una firma criptográfica. Esta ruta calcula esa firma en el backend (donde está el secreto) y la devuelve al frontend junto con la referencia.

💡Fórmula oficial de Wompi:
SHA256(reference + amountInCents + currency + WOMPI_INTEGRITY_SECRET)
El Widget verifica esa firma antes de mostrar el formulario de tarjeta. Si no coincide → el Widget no aparece.
⚠️Este archivo es nuevo — se crea, no se actualiza. Pero 3 detalles del código cambian frente a una guía Wompi genérica, porque así está construido este proyecto en particular:

const express        = require('express');
const crypto         = require('crypto');   // nativo de Node
const Orden          = require('../models/Orden');
const Transaccion    = require('../models/Transaccion');
const verificarToken = require('../middleware/auth');
const router         = express.Router();

// POST /api/pagos/firma — generar firma de integridad para el Widget de Wompi
// Valida el carrito, genera una referencia única, calcula la firma SHA256
// y guarda los datos del pedido en Transaccion con estado PENDING.
router.post('/firma', verificarToken, async (req, res) => {
  const { productos, total } = req.body;

  if (!productos?.length || !total) {
    return res.status(400).json({ error: 'Carrito vacío o total inválido' });
  }

  try {
    // 1. Referencia única para esta transacción
    const reference = `TS-${Date.now()}-${Math.floor(1000 + Math.random() * 9000)}`;
    const amountInCents = Math.round(total * 100);
    const currency = 'COP';

    // 2. Calcular firma de integridad — fórmula oficial Wompi
    const raw = `${reference}${amountInCents}${currency}${process.env.WOMPI_INTEGRITY_SECRET}`;
    const signature = crypto.createHash('sha256').update(raw).digest('hex');

    // 3. Guardar datos del carrito en la Transaccion con estado PENDING
    // ⚠️ req.usuario.id (no ._id) — así lo expone el middleware real de este proyecto
    await Transaccion.create({
      wompiReference: reference, amountInCents, currency,
      status: 'PENDING',
      pendingOrderData: { usuario: req.usuario.id, productos, total }
    });

    // 4. Devolver datos al Widget de Wompi en el frontend
    res.json({
      reference, amountInCents, currency, signature,
      publicKey: process.env.WOMPI_PUBLIC_KEY
    });

  } catch (err) {
    console.error(err);
    res.status(500).json({ error: 'Error al generar la firma' });
  }
});

// ── GET /api/pagos/estado/:reference ─────────────────────
// El frontend hace polling a esta ruta cada 3 segundos para saber si el
// pago fue aprobado. No necesita webhook ni ngrok en desarrollo.
router.get('/estado/:reference', verificarToken, async (req, res) => {
  try {
    const tx = await Transaccion.findOne({ wompiReference: req.params.reference });
    if (!tx) return res.status(404).json({ error: 'Transacción no encontrada' });
    res.json({ status: tx.status, ordenId: tx.orden ?? null });
  } catch (err) {
    res.status(500).json({ error: 'Error interno' });
  }

    .update(`${props}${timestamp}${secret}`).digest('hex');

  const transaccion = await Transaccion.findOne({ wompiReference: tx.reference });
  if (!transaccion) return res.status(404).json({ error: 'Transacción no encontrada' });

  transaccion.status = tx.status;
  transaccion.wompiTransactionId = tx.id;

  if (tx.status === 'APPROVED' && !transaccion.orden) {
    const pod = transaccion.pendingOrderData;
    const orden = await Orden.create({
      usuario: pod.usuario, productos: pod.productos, total: pod.total,
      wompiTransactionId: tx.id, wompiReference: tx.reference,
      estado: 'PAGO_CONFIRMADO'
    });
    transaccion.orden = orden._id;
  }

  await transaccion.save();
  res.json({ ok: true });

💡¿Qué es polling?
Polling = el frontend pregunta cada X segundos "¿ya cambió algo?". Es la técnica más sencilla y funciona perfecto en desarrollo local sin infraestructura extra.
⚠️La pieza que faltaba: ¿quién actualiza la Transaccion en MongoDB cuando Wompi aprueba el pago?
En producción → el webhook de Wompi (necesita una URL pública). En desarrollo sin ngrok, nadie llama al webhook, así que sin arreglar esto la Transaccion se queda en PENDING para siempre. La solución: mientras el endpoint de polling vea PENDING, el propio backend le pregunta a la API pública de Wompi por esa referencia, y si ya hay un resultado, actualiza la Transaccion ahí mismo — antes de responder al frontend.

// Máximo 20 intentos (60 segundos) antes de mostrar timeout.
function iniciarPolling(reference, token, mensaje) {
  btnConfirmar.textContent = 'Confirmando pago...';
  let intentos = 0;

  pollingInterval = setInterval(async function() {
    intentos++;
    if (intentos > 20) {
      clearInterval(pollingInterval);
      mostrarResultadoPago('TIMEOUT', mensaje);
      return;
    }

    try {
      const r = await fetch('http://localhost:3000/api/pagos/estado/' + reference, {
        headers: { 'Authorization': 'Bearer ' + token }
      });
      const data = await r.json();

      if (data.status !== 'PENDING') {
        clearInterval(pollingInterval);
        mostrarResultadoPago(data.status, mensaje);
      }
    } catch (e) {
      console.error('Polling error', e);
    }
  }, 3000);
}

🐛 Bug: el total se calcula mal con precios de más de 1 millón

Dónde está: frontend/js/main.js, dentro del listener de #btn-confirmar (bloque // ===== S18A: CHECKOUT — PAGAR CON WOMPI =====), la línea que calcula total.

Por qué pasa: el precio del producto viene como texto en formato colombiano, ej. "$8.999.000". Los puntos ahí son separadores de miles, no decimales. El código original solo quitaba el símbolo $ y dejaba los puntos:

frontend/js/main.js — código con el bugantes
const numero = parseFloat(item.precio.replace(/[^0-9.]/g, '')) || 0;

item.precio.replace(/[^0-9.]/g, '') convierte "$8.999.000" en el string "8.999.000" (le quita el $ pero conserva los 3 puntos). Luego parseFloat("8.999.000") lee solo hasta el primer punto como si fuera un decimal, e ignora el resto — el resultado es 8.999, no 8 millones. Con productos de menos de $1.000.000 (un solo punto o ninguno) el bug no se nota, por eso pasa desapercibido hasta probar con algo caro como el MacBook Pro M3.

Cómo arreglarlo: en frontend/js/main.js, buscar esa línea exacta y quitar el punto (.) de la lista de caracteres permitidos en el regex, para que también elimine los separadores de miles:

frontend/js/main.jscorregido
// Antes (con bug):
const numero = parseFloat(item.precio.replace(/[^0-9.]/g, '')) || 0;

// Después (corregido):
const numero = parseFloat(item.precio.replace(/[^0-9]/g, '')) || 0;

Con /[^0-9]/g (sin el . dentro de los corchetes) se eliminan todos los caracteres que no sean dígitos — el $ y los 3 puntos — dejando "8999000", que parseFloat lee correctamente como 8.999.000.

✅Cómo verificar que quedó bien: agregar al carrito un producto de más de $1.000.000 (ej. MacBook Pro M3 a $8.999.000), hacer clic en "💳 Pagar con Wompi", y confirmar que el Widget muestra el monto completo, no un número de una sola cifra.
🐛 Bug: la Transaccion se queda en PENDING aunque el pago sí fue aprobado

Dónde está: backend/routes/pago.js, dentro de GET /estado/:reference (el endpoint de polling), en el fetch que consulta la API de Wompi.

Por qué pasa: ese fetch usaba WOMPI_PUBLIC_KEY en el header Authorization:

backend/routes/pago.js — código con el bugantes
const wompiRes = await fetch(
  `https://sandbox.wompi.co/v1/transactions?reference=${tx.wompiReference}`,
  { headers: { Authorization: `Bearer ${process.env.WOMPI_PUBLIC_KEY}` } }
);

El endpoint GET /v1/transactions de Wompi (consultar transacciones por referencia) requiere la llave privada (WOMPI_PRIVATE_KEY, prefijo prv_test_...), no la pública. La llave pública solo sirve para abrir el Widget en el frontend. Como fetch no lanza excepción cuando la respuesta HTTP es un error (401/403) — solo la lanza si falla la red — el catch del endpoint nunca se activaba, el error quedaba en silencio, y la Transaccion se quedaba en PENDING para siempre aunque Wompi sí hubiera aprobado el pago.

Cómo arreglarlo:

  1. En el panel de Wompi (Sandbox → Desarrolladores → Llaves de API), copiar la "Llave privada" (prefijo prv_test_...).
  2. Agregarla a backend/.env: WOMPI_PRIVATE_KEY=prv_test_tu_llave_privada_aqui
  3. En backend/routes/pago.js, cambiar la llave usada en el fetch y agregar verificación de errores HTTP:
backend/routes/pago.jscorregido
// Antes (con bug):
const wompiRes = await fetch(
  `https://sandbox.wompi.co/v1/transactions?reference=${tx.wompiReference}`,
  { headers: { Authorization: `Bearer ${process.env.WOMPI_PUBLIC_KEY}` } }
);
const wompiJson = await wompiRes.json();
const wompiTx = wompiJson.data?.[0];

if (wompiTx && wompiTx.status !== 'PENDING') {
  await confirmarAprobado(tx, wompiTx);
}

// Después (corregido):
const wompiRes = await fetch(
  `https://sandbox.wompi.co/v1/transactions?reference=${tx.wompiReference}`,
  { headers: { Authorization: `Bearer ${process.env.WOMPI_PRIVATE_KEY}` } }
);

if (!wompiRes.ok) {
  console.error('Error consultando Wompi:', wompiRes.status, await wompiRes.text());
} else {
  const wompiJson = await wompiRes.json();
  const wompiTx = wompiJson.data?.[0];

  if (wompiTx && wompiTx.status !== 'PENDING') {
    await confirmarAprobado(tx, wompiTx);
  }
}
✅Cómo verificar que quedó bien: hacer un pago de prueba con la tarjeta 4242 4242 4242 4242. En la consola del backend, si algo sigue mal, ahora se verá el error real de Wompi en vez de fallar en silencio. En MongoDB Compass, la colección transaccions debe mostrar status: "APPROVED" y el campo orden con un ObjectId asignado — no PENDING.
🗑️Limpiar datos de prueba: las Transacciones que quedaron en PENDING por este bug (antes del fix) son basura de pruebas fallidas. Abrir MongoDB Compass, ir a la colección transaccions, buscar el documento por su wompiReference, y eliminarlo con "Delete document" — no hace falta un script, es un borrado manual puntual.
